#version 430 core

const int CSMLevels = 4;

layout(std140, binding = 1) uniform Sun
{
	vec3 sunDir;
	vec3 sunAmbient;
	vec3 sunDiffuseAndSpecular;
	vec3 skyColor;
};
layout(std140, binding = 2) uniform Shadow
{
	mat4 shadowMat[CSMLevels];
	mat4 shadowTexMat[CSMLevels];
};

uniform float fogDensity;

layout(binding = 0) uniform sampler2D texture0;
layout(binding = 1) uniform sampler2DArrayShadow shadowTex;
layout(binding = 2) uniform sampler2DArray shadow_tex_depth;

in vec3 modelPos;
in vec3 modelPosView;
in vec3 aNormal;
in vec2 aTexCoord;

out vec3 fragColor;

void main()
{
	float view_dist = length(modelPosView);
	vec3 color = texture(texture0, aTexCoord).rgb;
	vec3 lighting = sunAmbient;
	vec3 normal = normalize(aNormal);
	float LdotN = dot(sunDir, normal);
	vec3 surface_normal = normalize(cross(dFdx(modelPos), dFdy(modelPos)));
	if(LdotN > 0 && dot(surface_normal, sunDir) > 0)
	{
		const float texPadding = 0.02;
		float shadow;
		int level;
		vec3 shadowCoord;
		for(level = 0; level < CSMLevels; level++)
		{
			shadowCoord = (shadowTexMat[level] * vec4(modelPos, 1.0)).xyz;
			if(shadowCoord.x > texPadding && shadowCoord.x < 1 - texPadding &&
				shadowCoord.y > texPadding && shadowCoord.y < 1 - texPadding)
			{
				break;
			}
		}
		if(level < CSMLevels)
		{
			shadow = 1;
			vec3 offset_x = vec3(shadowTexMat[level][0].x,shadowTexMat[level][1].x,shadowTexMat[level][2].x);
			vec3 offset_y = vec3(shadowTexMat[level][0].y,shadowTexMat[level][1].y,shadowTexMat[level][2].y);
			vec3 shadow_mat_z = vec3(shadowTexMat[level][0].z,shadowTexMat[level][1].z,shadowTexMat[level][2].z);
			float shadow_space_x = length(offset_x);
			float shadow_space_y = length(offset_y);
			offset_x /= shadow_space_x;
			offset_y /= shadow_space_y;
			float dzdx, dzdy;
			vec3 vx = cross(offset_y, surface_normal);
			dzdx = dot(vx, shadow_mat_z) / dot(offset_x, vx);
			vec3 vy = cross(offset_x, surface_normal);
			dzdy = dot(vy, shadow_mat_z) / dot(offset_y, vy);
			const float blocker_sample_per_side = 5.0;
			const float blocker_search_radius = 100;
			int cnt = 0;
			float block_dist = 0;
			for(float i = -blocker_search_radius + blocker_search_radius / blocker_sample_per_side; i < blocker_search_radius; i += 2 * blocker_search_radius / blocker_sample_per_side)
			{
				for (float j = -blocker_search_radius + blocker_search_radius / blocker_sample_per_side; j < blocker_search_radius; j += 2 * blocker_search_radius / blocker_sample_per_side)
				{
					if(i * i + j * j > blocker_search_radius * blocker_search_radius)
					{
						continue;
					}
					float offset_z = min(0, i * dzdx + j * dzdy);
					vec2 coord = shadowCoord.xy + vec2(i * shadow_space_x, j * shadow_space_y);
					float dist = shadowCoord.z -  texture(shadow_tex_depth, vec3(coord.xy, level)).x;
					if(dist > -offset_z)
					{
						block_dist += dist;
						cnt++;
					}
				}
			}
			if(cnt > 0)
			{
				block_dist /= length(shadow_mat_z) * cnt;
				shadow = 0;
				const float sun_radius_dist_ratio = 1.0 / 214;
				float penumbra_radius = sun_radius_dist_ratio * block_dist;
				float sample_per_side = max(2, floor(1000 * penumbra_radius / view_dist));
				int sample_cnt = 0;
				float random_angle = fract(sin(dot(gl_FragCoord.xy * 12.9898, vec2(78.233, 45.164))) * 43758.5453) * 6.283185;
				float cosA = cos(random_angle);
				float sinA = sin(random_angle);
				mat2 rotMat = mat2(cosA, -sinA, sinA, cosA);
				for(float i = -penumbra_radius + penumbra_radius / sample_per_side; i < penumbra_radius; i += 2 * penumbra_radius / sample_per_side)
				{
					for (float j = -penumbra_radius + penumbra_radius / sample_per_side; j < penumbra_radius; j += 2 * penumbra_radius / sample_per_side)
					{
						if(i * i + j * j > penumbra_radius * penumbra_radius)
						{
							continue;
						}
						vec2 rotatedOffset = rotMat * vec2(i, j);
						float offset_z = min(0, rotatedOffset.x * dzdx +  rotatedOffset.y * dzdy);
						vec2 coord = shadowCoord.xy + vec2(rotatedOffset.x * shadow_space_x, rotatedOffset.y * shadow_space_y);
						shadow += texture(shadowTex, vec4(coord.xy, level, shadowCoord.z + offset_z)).x;
						sample_cnt++;
					}
				}
				shadow /= sample_cnt;
			}
			/*
			shadow = 0;
			vec3 offsetX = mat3(shadowTexMat[level]) * dFdx(modelPos);
			vec3 offsetY = mat3(shadowTexMat[level]) * dFdy(modelPos);
			float maxOffsetX = abs(offsetX.x) + abs(offsetY.x);
			if(maxOffsetX > texPadding)
			{
				offsetX *= texPadding / maxOffsetX;
				offsetY *= texPadding / maxOffsetX;
			}
			float maxOffsetY = abs(offsetX.y) + abs(offsetY.y);
			if(maxOffsetY > texPadding)
			{
				offsetX *= texPadding / maxOffsetY;
				offsetY *= texPadding / maxOffsetY;
			}
			const vec2 offsets[8] = { { -0.37, 0.37 }, { 0, 0.25 }, { 0.37, 0.37 }, { -0.25, 0 }, { 0.25, 0 }, { -0.37, -0.37 }, { 0, -0.25 }, { 0.37, -0.37 } };
			for(int i = 0; i < 8; i++)
			{
				vec3 coord = shadowCoord + offsets[i].x * offsetX + offsets[i].y * offsetY;
				shadow += texture(shadowTex, vec4(coord.xy, level, coord.z)).x;
			}
			shadow /= 8;
			*/
		}
		else
		{
			shadow = 1;
		}
		lighting += shadow * LdotN * sunDiffuseAndSpecular;
	}
	color *= lighting;
	
	float temp = fogDensity * max(0, view_dist - 30000);
	float fogFactor = exp(-temp * temp);
	fragColor = fogFactor * color + (1.0 - fogFactor) * skyColor;
}
