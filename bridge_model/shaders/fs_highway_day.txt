#include "shadow_day.glslh"

layout(std140, binding = 1) uniform Sun
{
	vec3 sunDir;
	vec3 sunAmbient;
	vec3 sunDiffuseAndSpecular;
	vec3 skyColor;
};

uniform float fogDensity;

layout(binding = 0) uniform sampler2D texture0;

in vec3 modelPos;
in vec3 modelPosView;
in vec3 aNormal;
in vec2 aTexCoord;

out vec3 fragColor;

void main()
{
	vec3 color = texture(texture0, aTexCoord).rgb;
	vec3 lighting = sunAmbient;
	vec3 normal = normalize(aNormal);
	float view_dist = length(modelPosView);
	float LdotN = dot(sunDir, normal);
	vec3 dpdx = dFdx(modelPos);
	vec3 dpdy = dFdy(modelPos);
	vec3 surface_normal = normalize(cross(dpdx, dpdy));
	if(LdotN > 1e-5 && dot(surface_normal, sunDir) > 1e-5)
	{
		float shadow = shadowPCSS(modelPos, dpdx, dpdy, surface_normal, LdotN);
		lighting += shadow * LdotN * sunDiffuseAndSpecular;
	}
	color *= lighting;
	float fogFactor;
	{
		float temp = fogDensity * max(0, view_dist - 30000);
		fogFactor = exp(-temp * temp);
	}
	fragColor = fogFactor * color + (1.0 - fogFactor) * skyColor;
}
