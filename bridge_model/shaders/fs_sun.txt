uniform float horizonY;

layout(pixel_center_integer) in vec4 gl_FragCoord;

layout(location = 0) out vec3 fragColor;
layout(location = 1) out vec3 brightColor;

layout(std140, binding = 1) uniform Sun
{
	vec3 sunDir;
	vec3 sunAmbient;
	vec3 sunDiffuseAndSpecular;
	vec3 skyColor;
};

void main()
{
	vec3 sunColor = 50.0 * sunDiffuseAndSpecular;
	if(gl_FragCoord.y < horizonY)
	{
		float sunPercentage = gl_FragCoord.y + 1 - horizonY;
		if(sunPercentage < 0)
		{
			discard;
		}
		sunColor *= sunPercentage;
		fragColor = sunColor + (1 - sunPercentage) * skyColor;
	}
	else
	{
		fragColor = sunColor;
	}
	brightColor = sunColor;
}
